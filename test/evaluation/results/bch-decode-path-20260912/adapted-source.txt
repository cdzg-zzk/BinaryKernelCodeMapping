426	/*
427	 * compute 2t syndromes of ecc polynomial, i.e. ecc(a^j) for j=1..2t
428	 */
429	static void compute_syndromes(struct bch_control *bch, uint32_t *ecc,
430				      unsigned int *syn)
431	{
432		int i, j, s;
433		unsigned int m;
434		uint32_t poly;
435		const int t = GF_T(bch);
436	
437		s = bch->ecc_bits;
438	
439		/* make sure extra bits in last ecc word are cleared */
440		m = ((unsigned int)s) & 31;
441		if (m)
442			ecc[s/32] &= ~((1u << (32-m))-1);
443		memset(syn, 0, 2*t*sizeof(*syn));
444	
445		/* compute v(a^j) for j=1 .. 2t-1 */
446		do {
447			poly = *ecc++;
448			s -= 32;
449			while (poly) {
450				i = deg(poly);
451				for (j = 0; j < 2*t; j += 2)
452					syn[j] ^= a_pow(bch, (j+1)*(i+s));
453	
454				poly ^= (1 << i);
455			}
456		} while (s > 0);
457	
458		/* v(a^(2j)) = v(a^j)^2 */
459		for (j = 0; j < t; j++)
460			syn[2*j+1] = gf_sqr(bch, syn[j]);
461	}

468	static int compute_error_locator_polynomial(struct bch_control *bch,
469						    const unsigned int *syn)
470	{
471		const unsigned int t = GF_T(bch);
472		const unsigned int n = GF_N(bch);
473		unsigned int i, j, tmp, l, pd = 1, d = syn[0];
474		struct gf_poly *elp = bch->elp;
475		struct gf_poly *pelp = bch->poly_2t[0];
476		struct gf_poly *elp_copy = bch->poly_2t[1];
477		int k, pp = -1;
478	
479		memset(pelp, 0, GF_POLY_SZ(2*t));
480		memset(elp, 0, GF_POLY_SZ(2*t));
481	
482		pelp->deg = 0;
483		pelp->c[0] = 1;
484		elp->deg = 0;
485		elp->c[0] = 1;
486	
487		/* use simplified binary Berlekamp-Massey algorithm */
488		for (i = 0; (i < t) && (elp->deg <= t); i++) {
489			if (d) {
490				k = 2*i-pp;
491				gf_poly_copy(elp_copy, elp);
492				/* e[i+1](X) = e[i](X)+di*dp^-1*X^2(i-p)*e[p](X) */
493				tmp = a_log(bch, d)+n-a_log(bch, pd);
494				for (j = 0; j <= pelp->deg; j++) {
495					if (pelp->c[j]) {
496						l = a_log(bch, pelp->c[j]);
497						elp->c[j+k] ^= a_pow(bch, tmp+l);
498					}
499				}
500				/* compute l[i+1] = max(l[i]->c[l[p]+2*(i-p]) */
501				tmp = pelp->deg+k;
502				if (tmp > elp->deg) {
503					elp->deg = tmp;
504					gf_poly_copy(pelp, elp_copy);
505					pd = d;
506					pp = 2*i;
507				}
508			}
509			/* di+1 = S(2i+3)+elp[i+1].1*S(2i+2)+...+elp[i+1].lS(2i+3-l) */
510			if (i < t-1) {
511				d = syn[2*i+2];
512				for (j = 1; j <= elp->deg; j++)
513					d ^= gf_mul(bch, elp->c[j], syn[2*i+2-j]);
514			}
515		}
516		dbg("elp=%s\n", gf_poly_str(elp));
517		return (elp->deg > t) ? -1 : (int)elp->deg;
518	}

649	/*
650	 * compute roots of a degree 2 polynomial over GF(2^m)
651	 */
652	static int find_poly_deg2_roots(struct bch_control *bch, struct gf_poly *poly,
653					unsigned int *roots)
654	{
655		int n = 0, i, l0, l1, l2;
656		unsigned int u, v, r;
657	
658		if (poly->c[0] && poly->c[1]) {
659	
660			l0 = bch->a_log_tab[poly->c[0]];
661			l1 = bch->a_log_tab[poly->c[1]];
662			l2 = bch->a_log_tab[poly->c[2]];
663	
664			/* using z=a/bX, transform aX^2+bX+c into z^2+z+u (u=ac/b^2) */
665			u = a_pow(bch, l0+l2+2*(GF_N(bch)-l1));
666			/*
667			 * let u = sum(li.a^i) i=0..m-1; then compute r = sum(li.xi):
668			 * r^2+r = sum(li.(xi^2+xi)) = sum(li.(a^i+Tr(a^i).a^k)) =
669			 * u + sum(li.Tr(a^i).a^k) = u+a^k.Tr(sum(li.a^i)) = u+a^k.Tr(u)
670			 * i.e. r and r+1 are roots iff Tr(u)=0
671			 */
672			r = 0;
673			v = u;
674			while (v) {
675				i = deg(v);
676				r ^= bch->xi_tab[i];
677				v ^= (1 << i);
678			}
679			/* verify root */
680			if ((gf_sqr(bch, r)^r) == u) {
681				/* reverse z=a/bX transformation and compute log(1/r) */
682				roots[n++] = modulo(bch, 2*GF_N(bch)-l1-
683						    bch->a_log_tab[r]+l2);
684				roots[n++] = modulo(bch, 2*GF_N(bch)-l1-
685						    bch->a_log_tab[r^1]+l2);
686			}
687		}
688		return n;
689	}

963	 * find roots of a polynomial, using BTZ algorithm; see the beginning of this
964	 * file for details
965	 */
966	static int find_poly_roots(struct bch_control *bch, unsigned int k,
967				   struct gf_poly *poly, unsigned int *roots)
968	{
969		int cnt;
970		struct gf_poly *f1, *f2;
971	
972		switch (poly->deg) {
973			/* handle low degree polynomials with ad hoc techniques */
974		case 1:
975			cnt = find_poly_deg1_roots(bch, poly, roots);
976			break;
977		case 2:
978			cnt = find_poly_deg2_roots(bch, poly, roots);
979			break;
980		case 3:
981			cnt = find_poly_deg3_roots(bch, poly, roots);
982			break;
983		case 4:
984			cnt = find_poly_deg4_roots(bch, poly, roots);
985			break;
986		default:
987			/* factor polynomial using Berlekamp Trace Algorithm (BTA) */
988			cnt = 0;
989			if (poly->deg && (k <= GF_M(bch))) {
990				factor_polynomial(bch, k, poly, &f1, &f2);
991				if (f1)
992					cnt += find_poly_roots(bch, k+1, f1, roots);
993				if (f2)
994					cnt += find_poly_roots(bch, k+1, f2, roots+cnt);
995			}
996			break;
997		}
998		return cnt;
999	}

1003	 * exhaustive root search (Chien) implementation - not used, included only for

1006	static int chien_search(struct bch_control *bch, unsigned int len,
1007				struct gf_poly *p, unsigned int *roots)
1008	{
1009		int m;
1010		unsigned int i, j, syn, syn0, count = 0;
1011		const unsigned int k = 8*len+bch->ecc_bits;
1012	
1013		/* use a log-based representation of polynomial */
1014		gf_poly_logrep(bch, p, bch->cache);
1015		bch->cache[p->deg] = 0;
1016		syn0 = gf_div(bch, p->c[0], p->c[p->deg]);
1017	
1018		for (i = GF_N(bch)-k+1; i <= GF_N(bch); i++) {
1019			/* compute elp(a^i) */
1020			for (j = 1, syn = syn0; j <= p->deg; j++) {
1021				m = bch->cache[j];
1022				if (m >= 0)
1023					syn ^= a_pow(bch, m+j*i);
1024			}
1025			if (syn == 0) {
1026				roots[count++] = GF_N(bch)-i;
1027				if (count == p->deg)
1028					break;
1029			}
1030		}
1031		return (count == p->deg) ? count : 0;
1032	}

1036	/**
1037	 * bch_decode - decode received codeword and find bit error locations
1038	 * @bch:      BCH control structure
1039	 * @data:     received data, ignored if @calc_ecc is provided
1040	 * @len:      data length in bytes, must always be provided
1041	 * @recv_ecc: received ecc, if NULL then assume it was XORed in @calc_ecc
1042	 * @calc_ecc: calculated ecc, if NULL then calc_ecc is computed from @data
1043	 * @syn:      hw computed syndrome data (if NULL, syndrome is calculated)
1044	 * @errloc:   output array of error locations
1045	 *
1046	 * Returns:
1047	 *  The number of errors found, or -EBADMSG if decoding failed, or -EINVAL if
1048	 *  invalid parameters were provided
1049	 *
1050	 * Depending on the available hw BCH support and the need to compute @calc_ecc
1051	 * separately (using bch_encode()), this function should be called with one of
1052	 * the following parameter configurations -
1053	 *
1054	 * by providing @data and @recv_ecc only:
1055	 *   bch_decode(@bch, @data, @len, @recv_ecc, NULL, NULL, @errloc)
1056	 *
1057	 * by providing @recv_ecc and @calc_ecc:
1058	 *   bch_decode(@bch, NULL, @len, @recv_ecc, @calc_ecc, NULL, @errloc)
1059	 *
1060	 * by providing ecc = recv_ecc XOR calc_ecc:
1061	 *   bch_decode(@bch, NULL, @len, NULL, ecc, NULL, @errloc)
1062	 *
1063	 * by providing syndrome results @syn:
1064	 *   bch_decode(@bch, NULL, @len, NULL, NULL, @syn, @errloc)
1065	 *
1066	 * Once bch_decode() has successfully returned with a positive value, error
1067	 * locations returned in array @errloc should be interpreted as follows -
1068	 *
1069	 * if (errloc[n] >= 8*len), then n-th error is located in ecc (no need for
1070	 * data correction)
1071	 *
1072	 * if (errloc[n] < 8*len), then n-th error is located in data and can be
1073	 * corrected with statement data[errloc[n]/8] ^= 1 << (errloc[n] % 8);
1074	 *
1075	 * Note that this function does not perform any data correction by itself, it
1076	 * merely indicates error locations.
1077	 */
1078	int bch_decode(struct bch_control *bch, const uint8_t *data, unsigned int len,
1079		       const uint8_t *recv_ecc, const uint8_t *calc_ecc,
1080		       const unsigned int *syn, unsigned int *errloc)
1081	{
1082		const unsigned int ecc_words = BCH_ECC_WORDS(bch);
1083		unsigned int nbits;
1084		int i, err, nroots;
1085		uint32_t sum;
1086	
1087		/* sanity check: make sure data length can be handled */
1088		if (8*len > (bch->n-bch->ecc_bits))
1089			return -EINVAL;
1090	
1091		/* if caller does not provide syndromes, compute them */
1092		if (!syn) {
1093			if (!calc_ecc) {
1094				/* compute received data ecc into an internal buffer */
1095				if (!data || !recv_ecc)
1096					return -EINVAL;
1097				bch_encode(bch, data, len, NULL);
1098			} else {
1099				/* load provided calculated ecc */
1100				load_ecc8(bch, bch->ecc_buf, calc_ecc);
1101			}
1102			/* load received ecc or assume it was XORed in calc_ecc */
1103			if (recv_ecc) {
1104				load_ecc8(bch, bch->ecc_buf2, recv_ecc);
1105				/* XOR received and calculated ecc */
1106				for (i = 0, sum = 0; i < (int)ecc_words; i++) {
1107					bch->ecc_buf[i] ^= bch->ecc_buf2[i];
1108					sum |= bch->ecc_buf[i];
1109				}
1110				if (!sum)
1111					/* no error found */
1112					return 0;
1113			}
1114			compute_syndromes(bch, bch->ecc_buf, bch->syn);
1115			syn = bch->syn;
1116		}
1117	
1118		err = compute_error_locator_polynomial(bch, syn);
1119		if (err > 0) {
1120			nroots = find_poly_roots(bch, 1, bch->elp, errloc);
1121			if (err != nroots)
1122				err = -1;
1123		}
1124		if (err > 0) {
1125			/* post-process raw error locations for easier correction */
1126			nbits = (len*8)+bch->ecc_bits;
1127			for (i = 0; i < err; i++) {
1128				if (errloc[i] >= nbits) {
1129					err = -1;
1130					break;
1131				}
1132				errloc[i] = nbits-1-errloc[i];
1133				if (!bch->swap_bits)
1134					errloc[i] = (errloc[i] & ~7) |
1135						    (7-(errloc[i] & 7));
1136			}
1137		}
1138		return (err >= 0) ? err : -EBADMSG;
1139	}
